cmake_minimum_required(VERSION 3.22)

set(SELF_FLIGHT_TARGET "host" CACHE STRING "Build target: host or mcu")
set_property(CACHE SELF_FLIGHT_TARGET PROPERTY STRINGS host mcu)
if(SELF_FLIGHT_TARGET STREQUAL "mcu")
    if(NOT CMAKE_TOOLCHAIN_FILE)
        set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/cmake/arm-gcc.cmake")
    endif()
elseif(NOT SELF_FLIGHT_TARGET STREQUAL "host")
    message(FATAL_ERROR "Unknown SELF_FLIGHT_TARGET='${SELF_FLIGHT_TARGET}'")
endif()
if(SELF_FLIGHT_TARGET STREQUAL "host" AND CMAKE_TOOLCHAIN_FILE)
    message(FATAL_ERROR "The host build does not accept a cross-compilation toolchain file. Use a clean native build directory.")
endif()

project(self_flight VERSION 0.2.0 LANGUAGES C CXX)
if(SELF_FLIGHT_TARGET STREQUAL "mcu")
    enable_language(ASM)
elseif(CMAKE_CROSSCOMPILING OR CMAKE_CXX_COMPILER MATCHES "arm-none-eabi")
    message(FATAL_ERROR "The host target requires a native compiler.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(flight_core STATIC
    flight/core/src/math.cpp
    flight/core/src/time.cpp
    flight/core/src/calibration.cpp
    flight/core/src/attitude.cpp
    flight/core/src/rate_feedback.cpp
    flight/core/src/imu_pipeline.cpp
    flight/core/src/flight_guard.cpp
)
add_library(self_flight::core ALIAS flight_core)
target_include_directories(flight_core PUBLIC flight/core/include)
target_compile_features(flight_core PUBLIC cxx_std_17)
add_library(bmi088 STATIC drivers/bmi088.cpp)
target_include_directories(bmi088 PUBLIC drivers)
target_link_libraries(bmi088 PUBLIC self_flight::core)
add_library(dshot STATIC drivers/dshot.cpp)
target_include_directories(dshot PUBLIC drivers)
target_link_libraries(dshot PUBLIC self_flight::core)
if(MSVC)
    target_compile_options(dshot PRIVATE /W4 /WX)
else()
    target_compile_options(dshot PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(sbus STATIC drivers/sbus.cpp)
target_include_directories(sbus PUBLIC drivers)
target_link_libraries(sbus PUBLIC self_flight::core)
if(MSVC)
    target_compile_options(sbus PRIVATE /W4 /WX)
else()
    target_compile_options(sbus PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MSVC)
    target_compile_options(bmi088 PRIVATE /W4 /WX)
else()
    target_compile_options(bmi088 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MSVC)
    target_compile_options(flight_core PRIVATE /W4 /WX /fp:precise)
else()
    target_compile_options(flight_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(SELF_FLIGHT_TARGET STREQUAL "mcu")
    if(BUILD_TESTING)
        message(FATAL_ERROR "Run native tests with a host preset.")
    endif()
    add_subdirectory(boards/micoair743v2_aio35)
else()
    include(CTest)
    if(BUILD_TESTING)
        add_subdirectory(tests/host)
    endif()
endif()
